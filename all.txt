Q15. Reader-Writer Problem using Mutex
Program:
#include<stdio.h>
#include<pthread.h>
#include<semaphore.h>

sem_t wrt;
pthread_mutex_t mutex;
int cnt=1;
int numreader=0;

void *writer(void *wno)
{
    sem_wait(&wrt);
    cnt=cnt*2;
    printf("Writer modified cnt to %d\n",cnt);
    sem_post(&wrt);
}

void *reader(void *rno)
{
    pthread_mutex_lock(&mutex);
    numreader++;
    if(numreader==1)
        sem_wait(&wrt);
    pthread_mutex_unlock(&mutex);

    printf("Reader reads cnt as %d\n",cnt);

    pthread_mutex_lock(&mutex);
    numreader--;
    if(numreader==0)
        sem_post(&wrt);
    pthread_mutex_unlock(&mutex);
}

int main()
{
    pthread_t read[5],write[5];
    pthread_mutex_init(&mutex,NULL);
    sem_init(&wrt,0,1);
    for(int i=0;i<5;i++)
    {
        pthread_create(&write[i],NULL,writer,NULL);
        pthread_create(&read[i],NULL,reader,NULL);
    }
    for(int i=0;i<5;i++)
    {
        pthread_join(write[i],NULL);
        pthread_join(read[i],NULL);
    }
}

Q17. Producer-Consumer Problem using Semaphore
Program:
#include<stdio.h>
#include<pthread.h>
#include<semaphore.h>

sem_t full,empty;
pthread_mutex_t mutex;
int buffer=0;

void *producer()
{
    sem_wait(&empty);
    pthread_mutex_lock(&mutex);
    buffer++;
    printf("Produced %d\n",buffer);
    pthread_mutex_unlock(&mutex);
    sem_post(&full);
}

void *consumer()
{
    sem_wait(&full);
    pthread_mutex_lock(&mutex);
    printf("Consumed %d\n",buffer);
    buffer--;
    pthread_mutex_unlock(&mutex);
    sem_post(&empty);
}

int main()
{
    pthread_t p,c;
    sem_init(&full,0,0);
    sem_init(&empty,0,1);
    pthread_mutex_init(&mutex,NULL);
    pthread_create(&p,NULL,producer,NULL);
    pthread_create(&c,NULL,consumer,NULL);
    pthread_join(p,NULL);
    pthread_join(c,NULL);
}

Q18. Producer-Consumer Problem using Mutex
Program:
#include<stdio.h>
#include<pthread.h>
#include<semaphore.h>

pthread_mutex_t mutex;
pthread_cond_t full,empty;
int buffer=0;
int capacity=5;

void *producer()
{
    pthread_mutex_lock(&mutex);
    while(buffer==capacity)
        pthread_cond_wait(&empty,&mutex);
    buffer++;
    printf("Produced %d\n",buffer);
    pthread_cond_signal(&full);
    pthread_mutex_unlock(&mutex);
}

void *consumer()
{
    pthread_mutex_lock(&mutex);
    while(buffer==0)
        pthread_cond_wait(&full,&mutex);
    printf("Consumed %d\n",buffer);
    buffer--;
    pthread_cond_signal(&empty);
    pthread_mutex_unlock(&mutex);
}

int main()
{
    pthread_t p,c;
    pthread_mutex_init(&mutex,NULL);
    pthread_cond_init(&full,NULL);
    pthread_cond_init(&empty,NULL);
    pthread_create(&p,NULL,producer,NULL);
    pthread_create(&c,NULL,consumer,NULL);
    pthread_join(p,NULL);
    pthread_join(c,NULL);
}

Q19. Dining Philosopher Problem using Semaphore
Program:
#include<stdio.h>
#include<pthread.h>
#include<semaphore.h>

sem_t chopstick[5];

void *philosopher(void *num)
{
    int id=*(int *)num;
    sem_wait(&chopstick[id]);
    sem_wait(&chopstick[(id+1)%5]);
    printf("Philosopher %d is eating\n",id);
    sem_post(&chopstick[id]);
    sem_post(&chopstick[(id+1)%5]);
}

int main()
{
    pthread_t p[5];
    int id[5];
    for(int i=0;i<5;i++)
        sem_init(&chopstick[i],0,1);
    for(int i=0;i<5;i++)
    {
        id[i]=i;
        pthread_create(&p[i],NULL,philosopher,&id[i]);
    }
    for(int i=0;i<5;i++)
        pthread_join(p[i],NULL);
}

SRTF Program:
#include<stdio.h>

int main()
{
    int n,at[10],bt[10],rt[10],ct[10],tat[10],wt[10];
    printf("Enter number of processes");
    scanf("%d",&n);
    for(int i=0;i<n;i++)
    {
        scanf("%d%d",&at[i],&bt[i]);
        rt[i]=bt[i];
    }
    int done=0,cur=0;
    while(done<n)
    {
        int s=-1,mn=9999;
        for(int i=0;i<n;i++)
            if(at[i]<=cur&&rt[i]>0&&rt[i]<mn)
            { mn=rt[i]; s=i; }
        if(s==-1){ cur++; continue; }
        rt[s]--;
        cur++;
        if(rt[s]==0)
        {
            ct[s]=cur;
            tat[s]=ct[s]-at[s];
            wt[s]=tat[s]-bt[s];
            done++;
        }
    }
    printf("P\tAT\tBT\tCT\tTAT\tWT\n");
    for(int i=0;i<n;i++)
        printf("%d\t%d\t%d\t%d\t%d\t%d\n",i,at[i],bt[i],ct[i],tat[i],wt[i]);
}

Round Robin Program:
#include<stdio.h>

int main()
{
    int n,tq,at[10],bt[10],rt[10],ct[10],tat[10],wt[10];
    printf("Enter number of processes and time quantum");
    scanf("%d%d",&n,&tq);
    for(int i=0;i<n;i++)
    {
        scanf("%d%d",&at[i],&bt[i]);
        rt[i]=bt[i];
    }
    int cur=0,done=0;
    while(done<n)
    {
        int fl=0;
        for(int i=0;i<n;i++)
        {
            if(rt[i]>0&&at[i]<=cur)
            {
                fl=1;
                if(rt[i]>tq){ cur+=tq; rt[i]-=tq; }
                else
                {
                    cur+=rt[i]; rt[i]=0;
                    ct[i]=cur; tat[i]=ct[i]-at[i];
                    wt[i]=tat[i]-bt[i]; done++;
                }
            }
        }
        if(!fl) cur++;
    }
    printf("P\tAT\tBT\tCT\tTAT\tWT\n");
    for(int i=0;i<n;i++)
        printf("%d\t%d\t%d\t%d\t%d\t%d\n",i,at[i],bt[i],ct[i],tat[i],wt[i]);
}

Q24. Banker's Algorithm

Program:
#include<stdio.h>

int main()
{
    int n,m;
    printf("Enter processes and resources");
    scanf("%d%d",&n,&m);
    int alloc[10][10],max[10][10],need[10][10],avail[10],finish[10],seq[10];
    for(int i=0;i<n;i++)
        for(int j=0;j<m;j++)
            scanf("%d",&alloc[i][j]);
    for(int i=0;i<n;i++)
        for(int j=0;j<m;j++)
            scanf("%d",&max[i][j]);
    for(int i=0;i<m;i++)
        scanf("%d",&avail[i]);
    for(int i=0;i<n;i++)
    {
        finish[i]=0;
        for(int j=0;j<m;j++)
            need[i][j]=max[i][j]-alloc[i][j];
    }
    int cnt=0;
    while(cnt<n)
    {
        int found=0;
        for(int i=0;i<n;i++)
        {
            if(finish[i]==0)
            {
                int ok=1;
                for(int j=0;j<m;j++)
                    if(need[i][j]>avail[j]){ ok=0; break; }
                if(ok)
                {
                    for(int j=0;j<m;j++)
                        avail[j]+=alloc[i][j];
                    finish[i]=1;
                    seq[cnt++]=i;
                    found=1;
                }
            }
        }
        if(!found){ printf("Unsafe State"); return 0; }
    }
    printf("Safe Sequence: ");
    for(int i=0;i<n;i++)
        printf("P%d ",seq[i]);
}

        for(int j=0;j<m;j++)
            scanf("%d",&req[i][j]);
    for(int i=0;i<m;i++)
        scanf("%d",&avail[i]);
    for(int i=0;i<n;i++)
    {
        finish[i]=1;
        for(int j=0;j<m;j++)
            if(alloc[i][j]!=0){ finish[i]=0; break; }
    }
    int found=1;
    while(found)
    {
        found=0;
        for(int i=0;i<n;i++)
        {
            if(finish[i]==0)
            {
                int ok=1;
                for(int j=0;j<m;j++)
                    if(req[i][j]>avail[j]){ ok=0; break; }
                if(ok)
                {
                    for(int j=0;j<m;j++)
                        avail[j]+=alloc[i][j];
                    finish[i]=1;
                    found=1;
                }
            }
        }
    }
    int deadlock=0;
    for(int i=0;i<n;i++)
        if(finish[i]==0){ printf("P%d is deadlocked\n",i); deadlock=1; }
    if(!deadlock)
        printf("No Deadlock");
}

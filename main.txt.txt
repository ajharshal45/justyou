# 6. Write a program to solve reader-writer problem using mutex.

```cpp id="q6"
#include<iostream>
#include<pthread.h>
#include<semaphore.h>

using namespace std;

sem_t wrt;
pthread_mutex_t mutex1;

int cnt=1;
int numreader=0;

void *writer(void *wno)
{
    sem_wait(&wrt);

    cnt=cnt*2;

    cout<<"Writer modified cnt to "<<cnt<<endl;

    sem_post(&wrt);
}

void *reader(void *rno)
{
    pthread_mutex_lock(&mutex1);

    numreader++;

    if(numreader==1)
        sem_wait(&wrt);

    pthread_mutex_unlock(&mutex1);

    cout<<"Reader reads cnt as "<<cnt<<endl;

    pthread_mutex_lock(&mutex1);

    numreader--;

    if(numreader==0)
        sem_post(&wrt);

    pthread_mutex_unlock(&mutex1);
}

int main()
{
    pthread_t read[5],write[5];

    pthread_mutex_init(&mutex1,NULL);

    sem_init(&wrt,0,1);

    for(int i=0;i<5;i++)
    {
        pthread_create(&write[i],NULL,writer,NULL);

        pthread_create(&read[i],NULL,reader,NULL);
    }

    for(int i=0;i<5;i++)
    {
        pthread_join(write[i],NULL);

        pthread_join(read[i],NULL);
    }
}
```

# 7. Write a program to solve producer-consumer problem using semaphore.

```cpp id="q7"
#include<iostream>
#include<pthread.h>
#include<semaphore.h>

using namespace std;

sem_t full,empty;
pthread_mutex_t mutex1;

int buffer=0;

void *producer()
{
    sem_wait(&empty);

    pthread_mutex_lock(&mutex1);

    buffer++;

    cout<<"Produced "<<buffer<<endl;

    pthread_mutex_unlock(&mutex1);

    sem_post(&full);
}

void *consumer()
{
    sem_wait(&full);

    pthread_mutex_lock(&mutex1);

    cout<<"Consumed "<<buffer<<endl;

    buffer--;

    pthread_mutex_unlock(&mutex1);

    sem_post(&empty);
}

int main()
{
    pthread_t p,c;

    sem_init(&full,0,0);

    sem_init(&empty,0,1);

    pthread_mutex_init(&mutex1,NULL);

    pthread_create(&p,NULL,producer,NULL);

    pthread_create(&c,NULL,consumer,NULL);

    pthread_join(p,NULL);

    pthread_join(c,NULL);
}
```

# 8. Write a program to solve dining philosopher problem using semaphore.

```cpp id="q8"
#include<iostream>
#include<pthread.h>
#include<semaphore.h>

using namespace std;

sem_t chopstick[5];

void *philosopher(void *num)
{
    int id=*(int *)num;

    sem_wait(&chopstick[id]);

    sem_wait(&chopstick[(id+1)%5]);

    cout<<"Philosopher "<<id<<" is eating"<<endl;

    sem_post(&chopstick[id]);

    sem_post(&chopstick[(id+1)%5]);
}

int main()
{
    pthread_t p[5];

    int id[5];

    for(int i=0;i<5;i++)
        sem_init(&chopstick[i],0,1);

    for(int i=0;i<5;i++)
    {
        id[i]=i;

        pthread_create(&p[i],NULL,philosopher,&id[i]);
    }

    for(int i=0;i<5;i++)
        pthread_join(p[i],NULL);
}
```

# 9. Draw the Gantt charts and write a C++ program to compute the finish time, turnaround time and waiting time for:

## a. First Come First Serve

## b. Shortest Job First (Preemptive)

## FCFS Program

```cpp id="q9fcfs"
#include<iostream>

using namespace std;

int main()
{
    int n;

    cout<<"Enter number of processes";
    cin>>n;

    int pid[20],at[20],bt[20];

    int ct[20],tat[20],wt[20];

    for(int i=0;i<n;i++)
    {
        pid[i]=i;

        cout<<"Enter Arrival Time and Burst Time";
        cin>>at[i]>>bt[i];
    }

    for(int i=0;i<n-1;i++)
    {
        for(int j=i+1;j<n;j++)
        {
            if(at[i]>at[j])
            {
                swap(at[i],at[j]);

                swap(bt[i],bt[j]);

                swap(pid[i],pid[j]);
            }
        }
    }

    ct[0]=at[0]+bt[0];

    for(int i=1;i<n;i++)
    {
        if(ct[i-1]<at[i])

            ct[i]=at[i]+bt[i];

        else

            ct[i]=ct[i-1]+bt[i];
    }

    for(int i=0;i<n;i++)
    {
        tat[i]=ct[i]-at[i];

        wt[i]=tat[i]-bt[i];
    }

    cout<<"PID\tAT\tBT\tCT\tTAT\tWT"<<endl;

    for(int i=0;i<n;i++)
    {
        cout<<pid[i]<<"\t"<<at[i]<<"\t"<<bt[i]<<"\t"<<ct[i]<<"\t"<<tat[i]<<"\t"<<wt[i]<<endl;
    }
}
```

## SJF Preemptive Program

```cpp id="q9sjf"
#include<iostream>

using namespace std;

int main()
{
    int n;

    cout<<"Enter number of processes";
    cin>>n;

    int at[20],bt[20],rt[20];

    for(int i=0;i<n;i++)
    {
        cout<<"Enter Arrival Time and Burst Time";
        cin>>at[i]>>bt[i];

        rt[i]=bt[i];
    }

    int complete=0,t=0,minm=9999;

    int shortest=0;

    bool check=false;

    while(complete!=n)
    {
        for(int j=0;j<n;j++)
        {
            if(at[j]<=t && rt[j]<minm && rt[j]>0)
            {
                minm=rt[j];

                shortest=j;

                check=true;
            }
        }

        if(check==false)
        {
            t++;
            continue;
        }

        rt[shortest]--;

        minm=rt[shortest];

        if(minm==0)
            minm=9999;

        if(rt[shortest]==0)
        {
            complete++;

            check=false;

            int finish=t+1;

            int wt=finish-bt[shortest]-at[shortest];

            int tat=bt[shortest]+wt;

            cout<<"P"<<shortest<<" WT="<<wt<<" TAT="<<tat<<endl;
        }

        t++;
    }
}
```

## Round Robin Program

```cpp id="q10rr"
#include<iostream>

using namespace std;

int main()
{
    int n,tq;

    cout<<"Enter number of processes";
    cin>>n;

    int at[20],bt[20],rt[20];

    for(int i=0;i<n;i++)
    {
        cout<<"Enter Arrival Time and Burst Time";
        cin>>at[i]>>bt[i];

        rt[i]=bt[i];
    }

    cout<<"Enter Time Quantum";
    cin>>tq;

    int time=0;

    while(1)
    {
        int done=1;

        for(int i=0;i<n;i++)
        {
            if(rt[i]>0 && at[i]<=time)
            {
                done=0;

                if(rt[i]>tq)
                {
                    time=time+tq;

                    rt[i]=rt[i]-tq;
                }
                else
                {
                    time=time+rt[i];

                    cout<<"P"<<i<<" completed at "<<time<<endl;

                    rt[i]=0;
                }
            }
        }

        if(done==1)
            break;
    }
}
```

# 5. Write a shell script which produces a report from the output of ls –l.

```bash id="q5shell"
#!/bin/bash

dir=0
file=0
link=0
size=0

for f in *
do
    if [ -d $f ]
    then
        echo "DIR $f/"

        dir=$((dir+1))

    elif [ -L $f ]
    then
        echo "LINK $f@"

        link=$((link+1))

    elif [ -f $f ]
    then
        echo "$f"

        file=$((file+1))

        s=$(stat -c%s "$f")

        size=$((size+s))
    fi
done

echo "Directories = $dir"

echo "Files = $file"

echo "Links = $link"

echo "Total File Size = $size"
```

```cpp id="q7cpp"
#include<iostream>
#include<pthread.h>

using namespace std;

pthread_mutex_t mutex1;
```

```cpp id="q8cpp"
int buffer=0;

void *producer()
{
    pthread_mutex_lock(&mutex1);

    buffer++;

    cout<<"Produced "<<buffer<<endl;

    pthread_mutex_unlock(&mutex1);
}

void *consumer()
{
    pthread_mutex_lock(&mutex1);

    cout<<"Consumed "<<buffer<<endl;

    buffer--;

    pthread_mutex_unlock(&mutex1);
}

int main()
{
    pthread_t p,c;

    pthread_mutex_init(&mutex1,NULL);

    pthread_create(&p,NULL,producer,NULL);

    pthread_create(&c,NULL,consumer,NULL);

    pthread_join(p,NULL);

    pthread_join(c,NULL);
}
```

```cpp id="fcfsdisk"
        total=total+abs(req[i]-head);

        head=req[i];
    }

    cout<<"Total Seek Time = "<<total<<endl;

    cout<<"Average Seek Time = "<<(float)total/n;
}
```

# 7. Write a program to implement following page replacement algorithms : FIFO, LRU

## FIFO Page Replacement

```cpp id="fifo"
#include<iostream>

using namespace std;

int main()
{
    int n,f;

    cin>>n>>f;

    int pages[50],frame[10];

    for(int i=0;i<n;i++)
    {
        cin>>pages[i];
    }

    for(int i=0;i<f;i++)
    {
        frame[i]=-1;
    }

    int index=0,fault=0;

    for(int i=0;i<n;i++)
    {
        int found=0;

        for(int j=0;j<f;j++)
        {
            if(frame[j]==pages[i])
                found=1;
        }

        if(found==0)
        {
            frame[index]=pages[i];

            index=(index+1)%f;

            fault++;
        }
    }

    cout<<"Page Faults = "<<fault;
}
```

## LRU Page Replacement

```cpp id="lru"
#include<iostream>

using namespace std;

int main()
{
    int n,f;

    cin>>n>>f;

    int pages[50],frame[10],time[10];

    for(int i=0;i<n;i++)
    {
        cin>>pages[i];
    }

    for(int i=0;i<f;i++)
    {
        frame[i]=-1;
    }

    int fault=0,counter=0;

    for(int i=0;i<n;i++)
    {
        int found=0;

        for(int j=0;j<f;j++)
        {
            if(frame[j]==pages[i])
            {
                counter++;

                time[j]=counter;

                found=1;
            }
        }

        if(found==0)
        {
            int pos=0;

            for(int j=1;j<f;j++)
            {
                if(time[j]<time[pos])
                    pos=j;
            }

            frame[pos]=pages[i];

            counter++;

            time[pos]=counter;

            fault++;
        }
    }

    cout<<"Page Faults = "<<fault;
}
```

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

char M[200][5];
char IR[5];
char R[5];

int IC;

void executeUserProgram()
{
    while(1)
    {
        // Fetch instruction
        strcpy(IR, M[IC]);

        IC++;

        // GD
        if(IR[0]=='G' && IR[1]=='D')
        {
            int loc = (IR[2]-'0')*10 + (IR[3]-'0');

            printf("Enter data: ");
            scanf("%s", M[loc]);
        }

        // PD
        else if(IR[0]=='P' && IR[1]=='D')
        {
            int loc = (IR[2]-'0')*10 + (IR[3]-'0');

        }

        // LR
        else if(IR[0]=='L' && IR[1]=='R')
        {
            int loc = (IR[2]-'0')*10 + (IR[3]-'0');

        // SR

        {
            int loc = (IR[2]-'0')*10 + (IR[3]-'0');

            strcpy(M[loc], R);

        // AD

        {
            int loc = (IR[2]-'0')*10 + (IR[3]-'0');

            int a = atoi(R);

            int b = atoi(M[loc]);

            int sum = a + b;

            sprintf(R, "%d", sum);
        }

        // H
        else if(IR[0]=='H')
        {
            break;
        }
    }
}

#include <iostream>
#include <fstream>
#include <string>

using namespace std;

// Main Memory
char M[400][4];

char IR[4];
char R[4];

int IC;
int SI;
int PI;

int PTR;

bool frameUsed[40];

fstream infile;
fstream outfile;

// ================= INIT =================
void INIT()
{
    for(int i=0;i<400;i++)
    {
        for(int j=0;j<4;j++)
        {
            M[i][j]=' ';
        }
    }

    for(int i=0;i<4;i++)
    {
        IR[i]=' ';
        R[i]=' ';
    }

    IC=0;
    SI=0;
    PI=0;

    PTR=0;

    for(int i=0;i<40;i++)
    {
        frameUsed[i]=false;
    }
}

// ================= ALLOCATE FRAME =================
int allocateFrame()
{
    for(int i=0;i<40;i++)
    {
        if(frameUsed[i]==false)
        {
            frameUsed[i]=true;

            return i;
        }
    }

    return -1;
}

// ================= ADDRESS MAP =================
int addressMap(int VA)
{
    int page = VA / 10;

    int offset = VA % 10;

    // PAGE NOT PRESENT
    if(M[PTR + page][0]==' ')
    {
        PI=3;

        return -1;
    }

    int frame =
        (M[PTR + page][2]-'0')*10 +
        (M[PTR + page][3]-'0');

    return frame*10 + offset;
}

    for(int i=0;i<10;i++)
    {
        for(int j=0;j<4;j++)
        {
            outfile << M[RA+i][j];
        }
    }

    outfile << endl;
}

void H()
{
    outfile << "\nPROGRAM TERMINATED\n";
}

{

    while(getline(infile,line))

        {

            START_EXECUTION();
        }

        // $END
        else if(line.substr(0,4)=="$END")
        {
            continue;
        }

        // Program Cards
        else
        {
            int frame = allocateFrame();

            int frameAddress = frame * 10;

            // Only first page loaded
            M[PTR + page][0]='1';

            M[PTR + page][2]=(frame/10)+'0';

                {

                    {
                        M[frameAddress+i][j]=line[k++];
                    }
                    else
                    {
                        M[frameAddress+i][j]=' ';
                    }
                }
            }
        }
    }

    infile.close();
    outfile.close();
}

Consider the virtual machine with one general purpose register of 4 bytes, instruction register of 4 bytes, instruction counter of 2 bytes & toggle register of a bit. The instruction set consists of 7 instructions of 4 bytes each. (GD, PD, LR, SR, CR, BT & H). Consider the main memory of 200 words. Write a program to print Hello World in the input file. Write the OS code which will execute this program and display errors. [Assume the program is already loaded in the main memory].

#include <iostream>
#include <fstream>
#include <string>

using namespace std;

char M[100][4];
char IR[4];
char R[4];

int IC;
int SI;
int PI;

fstream infile;
fstream outfile;

// ================= INIT =================
void INIT()
{
    for(int i=0;i<100;i++)
    {
        for(int j=0;j<4;j++)
        {
            M[i][j]=' ';
        }
    }

    for(int i=0;i<4;i++)
    {
        IR[i]=' ';
        R[i]=' ';
    }

    IC=0;
    SI=0;
    PI=0;
}

// ================= READ =================
void READ()
{
    string line;

    getline(infile,line);

    int loc =
        (IR[2]-'0')*10 +
        (IR[3]-'0');

    int k=0;

    for(int i=0;i<10;i++)
    {
        for(int j=0;j<4;j++)
        {
            if(k < line.length())
            {
                M[loc+i][j]=line[k++];
            }
            else
            {
                M[loc+i][j]=' ';
            }
        }
    }
}

// ================= WRITE =================
void WRITE()
{
    int loc =
        (IR[2]-'0')*10 +
        (IR[3]-'0');

    for(int i=0;i<10;i++)
    {
        for(int j=0;j<4;j++)
        {
            outfile << M[loc+i][j];
        }
    }

    outfile << endl;
}

// ================= TERMINATE =================
void TERMINATE()
{
    outfile << "\nProgram Terminated\n";
}

// ================= MOS =================
void MOS()
{
    // Service Interrupts
    if(SI==1)
    {
        READ();
    }

    else if(SI==2)
    {
        WRITE();
    }

    else if(SI==3)
    {
        TERMINATE();
    }

    // Program Interrupts
    if(PI==1)
    {
        outfile << "ERROR : OPERATION CODE ERROR\n";
    }

    else if(PI==2)
    {
        outfile << "ERROR : OPERAND ERROR\n";
    }

    SI=0;
    PI=0;
}

// ================= EXECUTE =================
void EXECUTE_USER_PROGRAM()
{
    while(true)
    {
        // FETCH
        for(int i=0;i<4;i++)
        {
            IR[i]=M[IC][i];
        }

        IC++;

        // ===== OPCODE VALIDATION =====
        bool validOpcode=false;

        if(
            (IR[0]=='G' && IR[1]=='D') ||
            (IR[0]=='P' && IR[1]=='D') ||
            (IR[0]=='L' && IR[1]=='R') ||
            (IR[0]=='S' && IR[1]=='R') ||
            (IR[0]=='C' && IR[1]=='R') ||
            (IR[0]=='B' && IR[1]=='T') ||
            (IR[0]=='H')
        )
        {
            validOpcode=true;
        }

        // OPCODE ERROR
        if(validOpcode==false)
        {
            PI=1;
            MOS();
            break;
        }

        // ===== OPERAND VALIDATION =====
        if(IR[0]!='H')
        {
            if(
                IR[2]<'0' || IR[2]>'9' ||
                IR[3]<'0' || IR[3]>'9'
            )
            {
                PI=2;
                MOS();
                break;
            }
        }

        // ===== EXECUTION =====

        // GD
        if(IR[0]=='G' && IR[1]=='D')
        {
            SI=1;
            MOS();
        }

        // PD
        else if(IR[0]=='P' && IR[1]=='D')
        {
            SI=2;
            MOS();
        }

        // H
        else if(IR[0]=='H')
        {
            SI=3;
            MOS();
            break;
        }
    }
}

}

The instruction set consists of 7 instructions of 4 bytes each. (GD, PD, LR, SR, CR, BT & H). Consider the main memory of 200 words. Write a program with time limit & line limit exceeded error. Write the OS code which will execute this program and display errors. [Assume the program is already loaded in the main memory].

#include<iostream>
using namespace std;

string M[100];

string IR;

int IC=0;

int TTC=0;

int TTL=3;

void TERMINATE()
{
    cout<<"TIME LIMIT EXCEEDED";
}

void EXECUTE()
{
    while(true)
    {
        IR=M[IC];

        IC++;

        TTC++;

        if(TTC > TTL)
        {
            TERMINATE();

            break;
        }

        if(IR[0]=='H')
            break;
    }
}

int main()
{
    M[0]="GD20";
    M[1]="PD20";
    M[2]="PD20";
    M[3]="PD20";
    M[4]="H   ";

    EXECUTE();

    return 0;
}
